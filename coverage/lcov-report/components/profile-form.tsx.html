"use client";import { useState } from "react";import { createClient } from "@/lib/supabase/client";type Values={first_name:string;last_name:string;email:string;phone:string};export default function ProfileForm({initial}:{initial:Values}){const [values,setValues]=useState(initial);const [message,setMessage]=useState("");const [error,setError]=useState("");const [busy,setBusy]=useState(false);function change(e:React.ChangeEvent<HTMLInputElement>){setValues({...values,[e.target.name]:e.target.value});}async function submit(e:React.FormEvent){e.preventDefault();setBusy(true);setMessage("");setError("");const supabase=createClient();const {error}=await supabase.from("ecom_users").update({first_name:values.first_name,last_name:values.last_name,phone:values.phone||null}).eq("id",(await supabase.auth.getUser()).data.user?.id);if(error)setError(error.message);else setMessage("Your details have been saved.");setBusy(false);}return <form className="profile-form" onSubmit={submit}><div className="form-grid"><label className="field">First name *<input name="first_name" required value={values.first_name} onChange={change}/></label><label className="field">Last name<input name="last_name" value={values.last_name} onChange={change}/></label><label className="field full-field">Email address<input value={values.email} disabled/><small>Email is managed by your sign-in provider.</small></label><label className="field full-field">Phone number<input name="phone" value={values.phone} onChange={change} autoComplete="tel"/></label></div>{message&&<p className="success-message">{message}</p>}{error&&<p className="form-error">{error}</p>}<button className="button button-primary" disabled={busy}>{busy?"Saving…":"Save changes"}</button></form>}
